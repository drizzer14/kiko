# Kiko Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Kiko foundation — domain models, local storage, manual entry, Monobank sync, price data, and simple screens.

**Architecture:** A local-only React Native app. op-sqlite stores relational data; Drizzle ORM types the schema and queries. Screens read through a custom reactive `useLiveQuery` hook built on op-sqlite reactive queries; every write runs inside a transaction. A functional pipeline pulls Monobank data; the `Money` value object holds currency amounts as integer minor units.

**Tech Stack:** React Native 0.87 (iOS), TypeScript, op-sqlite, Drizzle ORM, react-native-unistyles v3, React Navigation (native stack), react-native-keychain, ts-pattern, fnts, Jest + React Native Testing Library.

**Spec:** `docs/superpowers/specs/2026-08-30-kiko-foundation-design.md`

## Global Constraints

- Currencies: `BTC`, `USD`, `EUR`, `UAH` only. Scale: BTC = 8 decimals; USD, EUR, UAH = 2.
- Money is never a float. Store and compute in integer minor units.
- Every database write runs inside `db.transaction()`. A write outside a transaction will not trigger reactive updates.
- The Monobank token lives in the iOS Keychain. Never write it to the database, logs, or the repository.
- Follow `ovpn-ui` code conventions, adapted to Biome: full names, no abbreviations; uppercase abbreviations keep case (JSON, IBAN, MCC, ID, PAN); `import type` for type-only imports.
- Functional by default; OOP for data models; procedural where it fits.
- Use `ts-pattern` for exhaustive mapping. Use `fnts` for functional composition.
- Every check runs through the harness. Do not weaken a check to get green. Fix the underlying issue.
- New dependencies respect the 7-day minimum age in `.npmrc`.
- The harness fast tier runs on each edit; the medium tier runs on Stop. Keep them green.

```ts
// src/currency/money.ts
import { type Currency, currencyScale } from './currency';

export class Money {
    private constructor(
        readonly currency: Currency,
        readonly minorUnits: number,
    ) {}

    static of(currency: Currency, minorUnits: number): Money {
        return new Money(currency, Math.trunc(minorUnits));
    }

    static fromMajor(currency: Currency, major: number): Money {
        const factor = 10 ** currencyScale[currency];
        return new Money(currency, Math.round(major * factor));
    }

    private assertSameCurrency(other: Money): void {
        if (other.currency !== this.currency) {
            throw new Error(`Currency mismatch: ${this.currency} vs ${other.currency}`);
        }
    }

    add(other: Money): Money {
        this.assertSameCurrency(other);
        return new Money(this.currency, this.minorUnits + other.minorUnits);
    }

    subtract(other: Money): Money {
        this.assertSameCurrency(other);
        return new Money(this.currency, this.minorUnits - other.minorUnits);
    }

    negate(): Money {
        return new Money(this.currency, -this.minorUnits);
    }

    isZero(): boolean {
        return this.minorUnits === 0;
    }

    equals(other: Money): boolean {
        return this.currency === other.currency && this.minorUnits === other.minorUnits;
    }
}
```

// src/currency/format.ts
import { currencyScale } from './currency';
import type { Money } from './money';

export const formatMoney = (money: Money, locale = 'en-US'): string => {
    const scale = currencyScale[money.currency];
    const major = money.minorUnits / 10 ** scale;
    const formatted = major.toLocaleString(locale, {
        minimumFractionDigits: scale,
        maximumFractionDigits: scale,
    });
    return `${formatted} ${money.currency}`;
};

// src/db/use-live-query.ts
import { useEffect, useMemo, useState } from 'react';
import { rawDatabase } from './client';

type SQLQuery = { toSQL(): { sql: string; params: unknown[] } };

export function useLiveQuery<T>(
    query: SQLQuery,
    tables: string[],
): { data: T[]; error?: Error } {
    const [data, setData] = useState<T[]>([]);
    const [error, setError] = useState<Error | undefined>();
    const { sql, params } = useMemo(() => query.toSQL(), [query]);
    const fireOn = useMemo(() => tables.map((table) => ({ table })), [tables]);

    useEffect(() => {
        try {
            const unsubscribe = rawDatabase.reactiveExecute({
                query: sql,
                arguments: params,
                fireOn,
                callback: (response: { rows: T[] }) => setData(response.rows ?? []),
            });
            return unsubscribe;
        } catch (caught) {
            setError(caught instanceof Error ? caught : new Error(String(caught)));
            return undefined;
        }
    }, [sql, params, fireOn]);

    return { data, error };
}
```

import type { MonobankClientInfo, MonobankStatementItem } from './monobank.types';

const base = 'https://api.monobank.ua';

const request = async <T>(
    path: string,
    token: string,
    fetchImpl: typeof fetch,
): Promise<T> => {
    const response = await fetchImpl(`${base}${path}`, { headers: { 'X-Token': token } });
    if (!response.ok) {
        throw new Error(`Monobank request failed: ${response.status}`);
    }
    return (await response.json()) as T;
};

export const fetchClientInfo = (

export const fetchStatement = (

    fromSeconds: number,
    toSeconds: number,
    fetchImpl: typeof fetch = fetch,
): Promise<MonobankStatementItem[]> =>
    request(`/personal/statement/${accountId}/${fromSeconds}/${toSeconds}`, token, fetchImpl);

// src/rates/conversion.ts
import { type Currency, currencyScale } from '../currency/currency';
import { Money } from '../currency/money';

export type RateTable = Record<string, number>; // key: `${base}:${quote}` -> major-unit rate

const rateKey = (base: Currency, quote: Currency) => `${base}:${quote}`;

export const convert = (money: Money, target: Currency, rates: RateTable): Money => {
    if (money.currency === target) return money;
    const rate = rates[rateKey(money.currency, target)];
    if (rate === undefined) {
        throw new Error(`Missing rate ${money.currency}->${target}`);
    }
    const sourceMajor = money.minorUnits / 10 ** currencyScale[money.currency];
    const targetMajor = sourceMajor * rate;
    return Money.fromMajor(target, targetMajor);
};

export const netWorth = (
    holdings: { currency: Currency; balanceMinorUnits: number }[],
    base: Currency,
    rates: RateTable,
): Money =>
    holdings.reduce(
        (sum, holding) =>
            sum.add(convert(Money.of(holding.currency, holding.balanceMinorUnits), base, rates)),
        Money.of(base, 0),
    );
```
